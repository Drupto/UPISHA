   448→/* ─── Animated Section Wrapper ─── */
   449→function AnimatedSection({
   450→  children,
   451→  className = '',
   452→  id = '',
   453→}: {
   454→  children: React.ReactNode
   455→  className?: string
   456→  id?: string
   457→}) {
   458→  const ref = useRef(null)
   459→  const isInView = useInView(ref, { once: true, margin: '-50px' })
   460→
   461→  return (
   462→    <motion.section
   463→      ref={ref}
   464→      id={id}
   465→      initial={{ opacity: 0, y: 40 }}
   466→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   467→      transition={{ duration: 0.7, ease: 'easeOut' }}
   468→      className={className}
   469→    >
   470→      {children}
   471→    </motion.section>
   472→  )
   473→}

   518→/* ─── Navbar ─── */
   519→function Navbar({
   520→  activeSection,
   521→  onNavClick,
   522→}: {
   523→  activeSection: string
   524→  onNavClick: (href: string) => void
   525→}) {
   526→  const [isScrolled, setIsScrolled] = useState(false)
   527→  const [isMobileOpen, setIsMobileOpen] = useState(false)
   528→
   529→  useEffect(() => {
   530→    const handleScroll = () => setIsScrolled(window.scrollY > 40)
   531→    window.addEventListener('scroll', handleScroll)
   532→    return () => window.removeEventListener('scroll', handleScroll)
   533→  }, [])
   534→
   535→  return (
   536→    <header
   537→      className={`sticky top-0 z-50 transition-all duration-300 ${
   538→        isScrolled
   539→          ? 'bg-white/95 backdrop-blur-md shadow-lg border-b border-gray-100'
   540→          : 'bg-white shadow-sm'
   541→      }`}
   542→    >
   543→      <div className="max-w-7xl mx-auto px-4">
   544→        <div className="flex items-center justify-between h-16 md:h-20">
   545→          {/* Logo */}
   546→          <a
   547→            href="#home"
   548→            onClick={(e) => {
   549→              e.preventDefault()
   550→              onNavClick('#home')
   551→            }}
   552→            className="flex items-center gap-3 shrink-0"
   553→          >
   554→            <div className="w-10 h-10 md:w-12 md:h-12 bg-upisha-teal rounded-lg flex items-center justify-center">
   555→              <Ear className="h-6 w-6 md:h-7 md:w-7 text-white" />
   556→            </div>
   557→            <div className="hidden sm:block">
   558→              <div className="font-bold text-upisha-navy text-sm md:text-base leading-tight">
   559→                UP ISHA
   560→              </div>
   561→              <div className="text-[10px] md:text-xs text-upisha-teal font-medium leading-tight">
   562→                Speech & Hearing Association
   563→              </div>
   564→            </div>
   565→          </a>
   566→
   567→          {/* Desktop Nav */}
   568→          <nav className="hidden lg:flex items-center gap-1">
   569→            {navLinks.map((link) => (
   570→              <a
   571→                key={link.href}
   572→                href={link.href}
   573→                onClick={(e) => {
   574→                  e.preventDefault()
   575→                  onNavClick(link.href)
   576→                }}
   577→                className={`px-3 py-2 rounded-md text-sm font-medium transition-all duration-200 ${
   578→                  activeSection === link.href.slice(1)
   579→                    ? 'text-upisha-teal bg-upisha-teal-light'
   580→                    : 'text-gray-600 hover:text-upisha-teal hover:bg-upisha-teal-light/50'
   581→                }`}
   582→              >
   583→                {link.label}
   584→              </a>
   585→            ))}
   586→          </nav>
   587→
   588→          {/* CTA + Theme Toggle + Mobile Toggle */}
   589→          <div className="flex items-center gap-2 md:gap-3">
   590→            <ThemeToggle />
   591→            <Button
   592→              className="hidden md:inline-flex bg-upisha-teal hover:bg-upisha-teal-dark text-white"
   593→              onClick={() => onNavClick('#join')}
   594→            >
   595→              <UserPlus className="h-4 w-4 mr-2" />
   596→              Join Now
   597→            </Button>
   598→            <button
   599→              className="lg:hidden p-2 rounded-md hover:bg-gray-100 dark:hover:bg-gray-800 transition-colors"
   600→              onClick={() => setIsMobileOpen(!isMobileOpen)}
   601→              aria-label="Toggle navigation menu"
   602→            >
   603→              {isMobileOpen ? <X className="h-6 w-6" /> : <Menu className="h-6 w-6" />}
   604→            </button>
   605→          </div>
   606→        </div>
   607→      </div>
   608→
   609→      {/* Mobile Nav */}
   610→      <AnimatePresence>
   611→        {isMobileOpen && (
   612→          <motion.div
   613→            initial={{ height: 0, opacity: 0 }}
   614→            animate={{ height: 'auto', opacity: 1 }}
   615→            exit={{ height: 0, opacity: 0 }}
   616→            transition={{ duration: 0.3 }}
